# rsrc v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rewrite rsrc as a TypeScript-native, hook-based React library for managing async resource state with type-safe cross-resource cache invalidation.

**Architecture:** Four exports (`CacheProvider`, `createCache`, `defineResource`, `useResource`). Cache stores promises keyed by `key(params) + serialized queries`. `defineResource` returns typed hooks that can be referenced in other resources' `invalidates` arrays. Actions auto-refetch after success and pipe opaque metadata through a provider-level callback channel.

**Tech Stack:** TypeScript, React 18+, pnpm, Vitest, @testing-library/react, tsup (build)

---

```

  types.ts            — All type definitions (ResourceConfig, ActionConfig, CacheEntry, etc.)
  cache.ts            — createCache factory + Cache class

  useResource.test.ts — Inline hook: mount fetch, state transitions, refresh, invalidate
  defineResource.test.ts — Typed hooks, cross-resource invalidation, actions, auto-refetch
```

// ---- Cache types ----

export interface CacheEntry {
  value: Promise<unknown>;
  lastResolved?: number;
}

export interface CacheStore {
  get(key: string): CacheEntry | undefined;
  set(key: string, entry: CacheEntry): void;
  delete(key: string): void;
  keys(): IterableIterator<string>;
  has(key: string): boolean;
}

export interface Cache {
  get(key: string): CacheEntry | undefined;
  set(key: string, entry: CacheEntry): void;
  delete(key: string): void;
  keys(): IterableIterator<string>;
  has(key: string): boolean;
  subscribe(listener: () => void): () => void;
  notify(): void;
  invalidateByKeyBase(keyBase: string): void;
}

export interface CreateCacheOptions {
  store?: CacheStore;
}

// ---- Cache handle for custom invalidation functions ----

export interface CacheHandle {
  invalidate(resource: ResourceDefinition<any, any>, params?: Record<string, unknown>): void;
}

// ---- Resource definition marker ----

export interface ResourceDefinition<TParams, TQueries = undefined> {
  __brand: "rsrc_resource";
  key: (params: TParams) => string;
}

// ---- Action types ----

export type InvalidateTarget<TParams> =
  | ResourceDefinition<any, any>
  | ((params: TParams, cache: CacheHandle) => void);

export interface ActionConfig<TParams, TBody = void, TResponse = unknown, TMeta = unknown> {
  fn: (params: TParams, body: TBody) => Promise<TResponse>;
  invalidates?: InvalidateTarget<TParams>[];
  meta?: TMeta;
}

// ---- Resource config ----

export type ActionsConfig<TParams> = Record<string, ActionConfig<TParams, any, any, any>>;

export interface ResourceConfig<
  TParams,
  TQueries,
  TData,
  TActions extends ActionsConfig<TParams>,
  TMeta = unknown,
> {
  key: (params: TParams) => string;
  get: (params: TParams, queries?: TQueries) => Promise<TData>;
  maxAge?: number;
  actions?: TActions;
}

// ---- Inline resource config (useResource) ----

export interface InlineResourceConfig<TData, TActions extends ActionsConfig<void>> {
  key: string;
  get: () => Promise<TData>;
  maxAge?: number;
  actions?: TActions;
}

// ---- Hook return ----

export interface ResourceState<TData> {
  pending: boolean;
  fulfilled: boolean;
  rejected: boolean;
  value: TData | null;
  reason: Error | null;
}

export type MappedActions<TActions extends ActionsConfig<any>> = {
  [K in keyof TActions]: TActions[K] extends ActionConfig<any, infer TBody, infer TResponse, any>
    ? (body: TBody) => Promise<TResponse>
    : never;
};

export interface UseResourceReturn<TData, TActions extends ActionsConfig<any>> {
  state: ResourceState<TData>;
  actions: MappedActions<TActions>;
  refresh: () => void;
  invalidate: () => void;
}

// ---- Provider types ----

export interface CacheProviderProps {
  cache: Cache;
  onActionSuccess?: (meta: unknown, result: unknown) => void;
  onActionError?: (meta: unknown, error: Error) => void;
  children: React.ReactNode;
}

import type { Cache, CacheEntry, CacheStore, CreateCacheOptions } from "./types";

export function createCache(options?: CreateCacheOptions): Cache {
  const store: CacheStore = (options?.store as CacheStore) ?? new Map<string, CacheEntry>();
  const listeners = new Set<() => void>();

  function notify() {
    listeners.forEach((fn) => fn());
  }

  function subscribe(listener: () => void): () => void {
    listeners.add(listener);
    return () => {
      listeners.delete(listener);
    };
  }

  function invalidateByKeyBase(keyBase: string): void {
    const toDelete: string[] = [];
    for (const key of store.keys()) {
      if (key === keyBase || key.startsWith(keyBase + "::")) {
        toDelete.push(key);
      }
    }
    if (toDelete.length === 0) return;
    toDelete.forEach((key) => store.delete(key));
    notify();
  }

  return {
    get: (key) => store.get(key),
    set: (key, entry) => {
      store.set(key, entry);
      notify();
    },
    delete: (key) => {
      store.delete(key);
      notify();
    },
    keys: () => store.keys(),
    has: (key) => store.has(key),
    subscribe,
    notify,
    invalidateByKeyBase,
  };
}
```

import { createContext, useContext, createElement } from "react";
import type { Cache } from "./types";

export interface CacheContextValue {
  cache: Cache;
  onActionSuccess?: (meta: unknown, result: unknown) => void;
  onActionError?: (meta: unknown, error: Error) => void;
}

const CacheContext = createContext<CacheContextValue | null>(null);

export function useCacheContext(): CacheContextValue | null {
  return useContext(CacheContext);
}

export interface CacheProviderProps {
  cache: Cache;
  onActionSuccess?: (meta: unknown, result: unknown) => void;
  onActionError?: (meta: unknown, error: Error) => void;
  children: React.ReactNode;
}

export function CacheProvider({ cache, onActionSuccess, onActionError, children }: CacheProviderProps) {
  return createElement(
    CacheContext.Provider,
    { value: { cache, onActionSuccess, onActionError } },
    children,
  );
}

import React from "react";
import { useResource } from "../src/useResource";
import { CacheProvider } from "../src/context";
import { createCache } from "../src/cache";
import type { Cache } from "../src/types";

let cache: Cache;

function wrapper({ children }: { children: React.ReactNode }) {
  return React.createElement(CacheProvider, { cache, children });
}

beforeEach(() => {
  cache = createCache();
});

describe("useResource", () => {
  it("fetches on mount and transitions to fulfilled", async () => {
    const get = vi.fn(() => Promise.resolve({ name: "test" }));

    const { result } = renderHook(
      () => useResource({ key: "/test", get }),
      { wrapper },
    );

    expect(result.current.state.pending).toBe(true);

    await waitFor(() => {
      expect(result.current.state.fulfilled).toBe(true);
    });

    expect(result.current.state.value).toEqual({ name: "test" });
    expect(result.current.state.pending).toBe(false);
    expect(result.current.state.rejected).toBe(false);
  });

  it("transitions to rejected on fetch error", async () => {
    const error = new Error("boom");
    const get = vi.fn(() => Promise.reject(error));

    const { result } = renderHook(
      () => useResource({ key: "/fail", get }),
      { wrapper },
    );

    await waitFor(() => {
      expect(result.current.state.rejected).toBe(true);
    });

    expect(result.current.state.reason).toBe(error);
    expect(result.current.state.fulfilled).toBe(false);
  });

  it("uses cache on second mount with same key", async () => {
    const get = vi.fn(() => Promise.resolve("data"));

    const { result, unmount } = renderHook(
      () => useResource({ key: "/cached", get }),
      { wrapper },
    );

    await waitFor(() => expect(result.current.state.fulfilled).toBe(true));
    unmount();

    const { result: result2 } = renderHook(
      () => useResource({ key: "/cached", get }),
      { wrapper },
    );

    await waitFor(() => expect(result2.current.state.fulfilled).toBe(true));
    expect(get).toHaveBeenCalledTimes(1);
  });

    let callCount = 0;
    const get = vi.fn(() => Promise.resolve(`data-${++callCount}`));

    const { result } = renderHook(
      () => useResource({ key: "/refresh", get }),
      { wrapper },
    );

    await waitFor(() => expect(result.current.state.fulfilled).toBe(true));
    expect(result.current.state.value).toBe("data-1");

    act(() => {
      result.current.refresh();
    });

    await waitFor(() => {
      expect(result.current.state.value).toBe("data-2");
    });

    expect(get).toHaveBeenCalledTimes(2);
  });

```typescript
import { useState, useEffect, useCallback, useRef, useMemo } from "react";
import { useCacheContext } from "./context";
import { createCache } from "./cache";
import type {
  Cache,
  CacheEntry,
  ResourceState,
  ActionsConfig,
  UseResourceReturn,
  MappedActions,
} from "./types";

const DEFAULT_MAX_AGE = 60;

export interface InlineConfig<TData, TActions extends ActionsConfig<void>> {
  key: string;
  get: () => Promise<TData>;
  maxAge?: number;
  actions?: TActions;
}

function buildCacheKey(key: string, queries?: Record<string, unknown>): string {
  if (!queries || Object.keys(queries).length === 0) return key;
  return `${key}::${JSON.stringify(queries)}`;
}

function isStale(entry: CacheEntry, maxAge: number): boolean {
  if (entry.lastResolved == null) return false;
  return entry.lastResolved + maxAge * 1000 < Date.now();
}

export function useResourceInternal<TData, TActions extends ActionsConfig<any>>(
  key: string,
  get: (...args: any[]) => Promise<TData>,
  getArgs: unknown[],
  maxAge: number,
  actions: TActions | undefined,
  actionParams: unknown,
): UseResourceReturn<TData, TActions> {
  const ctx = useCacheContext();
  const ephemeralCache = useRef<Cache | null>(null);

  if (!ctx && !ephemeralCache.current) {
    if (typeof console !== "undefined") {
      console.warn("rsrc: No CacheProvider found. Falling back to ephemeral cache.");
    }
    ephemeralCache.current = createCache();
  }

  const cache = ctx?.cache ?? ephemeralCache.current!;

  const [state, setState] = useState<ResourceState<TData>>({
    pending: true,
    fulfilled: false,
    rejected: false,
    value: null,
    reason: null,
  });

  const promiseRef = useRef<Promise<TData> | null>(null);
  const mountedRef = useRef(true);

  useEffect(() => {
    mountedRef.current = true;
    return () => {
      mountedRef.current = false;
    };
  }, []);

  const read = useCallback(() => {
    const cached = cache.get(key);

    if (cached && !isStale(cached, maxAge)) {
      const promise = cached.value as Promise<TData>;
      promiseRef.current = promise;

      setState((prev) => ({ ...prev, pending: true }));

      promise.then(
        (value) => {
          if (mountedRef.current && promiseRef.current === promise) {
            setState({ pending: false, fulfilled: true, rejected: false, value, reason: null });
          }
        },
        (reason) => {
          if (mountedRef.current && promiseRef.current === promise) {
            setState({ pending: false, fulfilled: false, rejected: true, value: null, reason });
          }
        },
      );
      return;
    }

    setState((prev) => ({ ...prev, pending: true }));

    const promise = get(...getArgs);
    promiseRef.current = promise;

    cache.set(key, { value: promise });

    promise.then(
      (value) => {
        const entry = cache.get(key);
        if (entry && entry.value === promise) {
          cache.set(key, { ...entry, lastResolved: Date.now() });
        }
        if (mountedRef.current && promiseRef.current === promise) {
          setState({ pending: false, fulfilled: true, rejected: false, value, reason: null });
        }
      },
      (reason) => {
        const entry = cache.get(key);
        if (entry && entry.value === promise) {
          cache.set(key, { ...entry, lastResolved: Date.now() });
        }
        if (mountedRef.current && promiseRef.current === promise) {
          setState({ pending: false, fulfilled: false, rejected: true, value: null, reason });
        }
      },
    );
  }, [cache, key, maxAge, get, ...getArgs]);

  // Initial fetch on mount + refetch when key changes
  useEffect(() => {
    read();
  }, [read]);

  // Subscribe to cache changes (from other resources invalidating us)
  useEffect(() => {
    return cache.subscribe(() => {
      if (!cache.has(key) && mountedRef.current) {
        read();
      }
    });
  }, [cache, key, read]);

  const refresh = useCallback(() => {
    cache.delete(key);
    read();
  }, [cache, key, read]);

  const invalidate = useCallback(() => {
    cache.delete(key);
  }, [cache, key]);

  const mappedActions = useMemo(() => {
    if (!actions) return {} as MappedActions<TActions>;

    const mapped: Record<string, (body: unknown) => Promise<unknown>> = {};

    for (const [name, actionConfig] of Object.entries(actions)) {
      mapped[name] = async (body: unknown) => {
        try {
          const result = await actionConfig.fn(actionParams, body);

          // Run invalidation
          if (actionConfig.invalidates) {
            for (const target of actionConfig.invalidates) {
              if (typeof target === "function") {
                target(actionParams as any, {
                  invalidate: (resource: any, params?: Record<string, unknown>) => {
                    const targetKey = resource.key(params ?? actionParams);
                    cache.invalidateByKeyBase(targetKey);
                  },
                });
              } else {
                const targetKey = target.key(actionParams as any);
                cache.invalidateByKeyBase(targetKey);
              }
            }
          }

          // Meta channel
          ctx?.onActionSuccess?.(actionConfig.meta, result);

          // Auto-refetch own GET
          cache.delete(key);
          read();

          return result;
        } catch (error) {
          ctx?.onActionError?.(actionConfig.meta, error as Error);
          throw error;
        }
      };
    }

    return mapped as MappedActions<TActions>;
  }, [actions, actionParams, cache, key, read, ctx]);

  return { state, actions: mappedActions, refresh, invalidate };
}

export function useResource<TData, TActions extends ActionsConfig<void> = Record<string, never>>(
  config: InlineConfig<TData, TActions>,
): UseResourceReturn<TData, TActions> {
  const cacheKey = buildCacheKey(config.key);
  return useResourceInternal<TData, TActions>(
    cacheKey,
    config.get,
    [],
    config.maxAge ?? DEFAULT_MAX_AGE,
    config.actions,
    undefined,
  );
}

import type {
  ActionsConfig,
  ResourceDefinition,
  UseResourceReturn,
} from "./types";
import { useResourceInternal } from "./useResource";

const DEFAULT_MAX_AGE = 60;

interface DefineResourceConfig<TParams, TQueries, TData, TActions extends ActionsConfig<TParams>> {
  key: (params: TParams) => string;
  get: (params: TParams, queries?: TQueries) => Promise<TData>;
  maxAge?: number;
  actions?: TActions;
}

function buildCacheKey(key: string, queries?: Record<string, unknown>): string {
  if (!queries || Object.keys(queries).length === 0) return key;
  return `${key}::${JSON.stringify(queries)}`;
}

export function defineResource<
  TParams,
  TQueries = undefined,
  TData = unknown,
  TActions extends ActionsConfig<TParams> = Record<string, never>,
>(
  config: DefineResourceConfig<TParams, TQueries, TData, TActions>,
): ResourceDefinition<TParams, TQueries> &
  ((params: TParams, queries?: TQueries) => UseResourceReturn<TData, TActions>) {
  const maxAge = config.maxAge ?? DEFAULT_MAX_AGE;

  const hook = (params: TParams, queries?: TQueries): UseResourceReturn<TData, TActions> => {
    const keyBase = config.key(params);
    const cacheKey = buildCacheKey(keyBase, queries as Record<string, unknown> | undefined);

    return useResourceInternal<TData, TActions>(
      cacheKey,
      config.get,
      queries !== undefined ? [params, queries] : [params],
      maxAge,
      config.actions,
      params,
    );
  };

  // Attach ResourceDefinition properties so this can be used in invalidates
  (hook as any).__brand = "rsrc_resource" as const;
  (hook as any).key = config.key;

  return hook as ResourceDefinition<TParams, TQueries> &
    ((params: TParams, queries?: TQueries) => UseResourceReturn<TData, TActions>);
}

# rsrc

React hooks for managing async resource state with type-safe cache invalidation.

## Install

```bash
pnpm add rsrc

## Quick Start

```typescript
import { CacheProvider, createCache, defineResource } from 'rsrc';

// 1. Create a cache and wrap your app
const cache = createCache();

function App() {
  return (
    <CacheProvider cache={cache}>
      <MyComponent />
    </CacheProvider>
  );
}

// 2. Define a resource
const useUser = defineResource({
  key: ({ userId }) => `/users/${userId}`,
  get: ({ userId }) => fetch(`/api/users/${userId}`).then(r => r.json()),
});

// 3. Use it
function MyComponent() {
  const { state } = useUser({ userId: '123' });

  if (state.pending) return <div>Loading...</div>;
  if (state.rejected) return <div>Error: {state.reason?.message}</div>;
  return <div>{state.value.name}</div>;
}
